# Package set for the generic Foundation TerminalOS core (spec §1 minimal). No specific
# device is assumed here — see profiles/$HARDWARE_PROFILE/packages.txt for
# anything a hardware profile adds on top (00-base-packages.sh installs both).
# Grouped by purpose. One package per line; # = comment.

# --- kernel (spec §1: linux-lts, pinned) ---
linux-lts
linux-lts-headers
linux-firmware
# CPU microcode is vendor-specific; a profile targeting an AMD device should
# add amd-ucode instead. intel-ucode is a reasonable generic default since
# it's a no-op on non-Intel CPUs, but move it to a profile if that stops being true.
intel-ucode

# --- console kiosk (spec §2: the kernel VT is the display; no GPU stack) ---
# There is deliberately NO compositor or graphical terminal here. The Hub is a
# curses app running directly on the kernel text console — any x86_64 machine
# that boots Linux gets a working screen, the way any PC that booted DOS got
# one from the BIOS. A hardware profile whose device glue genuinely needs a
# display stack (e.g. zenbook-duo-2024's dual-panel Wayland/wlr-randr glue)
# adds cage/kitty/wlr-randr/seatd in ITS packages.txt — see docs/PROFILES.md.
terminus-font   # bitmap console faces for the phosphor look + text-size setting
                # (feedback #1: ter-v16b/v24b/v32b; default ter-v32b, the 2× face)

# --- boot (spec §3) ---
grub
plymouth        # text theme only; real boot log stays visible

# --- generic hardware helpers (harmless no-ops if the machine lacks the device) ---
power-profiles-daemon  # backs the Functions Control power-profile toggle
bluez                   # Bluetooth peripherals in general
bluez-utils

# --- the Hub + Frank runtime (spec §4, §6) ---
python
# (foundationhub and frankd are pure-stdlib; no pip deps required for offline mode)
polkit          # pkexec path for foundationhub-account (login registration, docs/USERS.md)
sudo            # used by install/05's isolation probes; previously assumed from the hand base install

# --- Programs area tools (spec §5) ---
# DIRECTION: all of these are TEMPORARY stand-ins to be replaced in-house
# (docs/OPEN-QUESTIONS.md §10). cmus and mpv are already dropped — media is
# one in-house player (foundationmedia), not two borrowed ones.
# foundationmedia BUILT (BUILD-QUEUE §4): media/foundationmedia/, installed by
# install/09-media.sh. Backend is the operator-approved hybrid: in-house core
# (stdlib WAV/AIFF decode + ctypes ALSA output via alsa-lib) with ffmpeg as an
# optional broad-format decode ENGINE under the in-house UI — a build tool,
# not a borrowed app. Without ffmpeg the player still runs, WAV/AIFF-only
# (that stdlib-only path is the Pocket8086-tier fallback).
alsa-lib        # libasound — foundationmedia's in-house ctypes PCM output
ffmpeg          # foundationmedia's optional decode engine (mp3/flac/ogg/…)
# neovim RETIRED (BUILD-QUEUE §1): the in-house notes-suite editor
# (hub/foundationhub/editor.py) is THE system editor now — one editor everywhere.
# ranger RETIRED (BUILD-QUEUE §2): the in-house file manager
# (hub/foundationhub/screens/files.py) is THE file manager now, scoped to the
# account's own quota'd data space.
# btop RETIRED (BUILD-QUEUE §3): the in-house system monitor
# (hub/foundationhub/screens/monitor.py + hub/foundationhub/sysinfo.py) reads /proc
# directly, stdlib only — view-only, no kill/renice.
networkmanager  # nmtui network config

# --- optional: AI features (spec §5, §6) need python-requests when a key is set ---
python-requests

# --- Recreation defaults (spec §5; in-house mandate — see OPEN-QUESTIONS §2/§10) ---
# nsnake/vitetris/ninvaders/2048/nudoku RETIRED (BUILD-QUEUE §5 item 1):
# Foundation Arcade (games/foundation_arcade/, installed by install/08-games.sh)
# is one in-house program replacing all five.
# gnuchess RETIRED (BUILD-QUEUE §5 item 2): Foundation Chess

nethack
